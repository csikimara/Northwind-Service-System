function escHtml(s) {
  return String(s || '')
    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function getCalendar() {
  var cals = CalendarApp.getCalendarsByName("Northwind Munka");
  return cals.length ? cals[0] : CalendarApp.getDefaultCalendar();
}

function getSheet(name) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  return ss.getSheetByName(name) || ss.insertSheet(name);
}

function validMode(mode) {
  return ['idopont', 'felmeres', 'telepites', 'karbantartas', 'javitas', 'egyeb'].indexOf(String(mode || '').toLowerCase()) > -1;
}

function folderByMode(mode) {
  mode = String(mode || '').toLowerCase();
  if (mode === 'telepites') return 'telepiteskepek';
  if (mode === 'felmeres') return 'munkafelmereskepek';
  if (mode === 'karbantartas') return 'karbantartaskepek';
  if (mode === 'javitas') return 'javitaskepek';
  return 'egyebkepek';
}

function getOrCreateFolder(name) {
  var it = DriveApp.getFoldersByName(name);
  return it.hasNext() ? it.next() : DriveApp.createFolder(name);
}

function frissitEmlekezteto(ugyfel, cim, email, tel, honap) {
  var sh = getSheet("Emlékeztetők");
  if (sh.getLastRow() === 0) {
    sh.appendRow(["Ügyfél", "Cím", "Email", "Telefon", "Következő Dátum", "Értesítve"]);
  }

  var nextDate = new Date();
  nextDate.setMonth(nextDate.getMonth() + (parseInt(honap, 10) || 12));
  var dateStr = Utilities.formatDate(nextDate, Session.getScriptTimeZone(), "yyyy-MM-dd");

  var data = sh.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    var rowName = String(data[i][0] || "");
    var rowEmail = String(data[i][2] || "");
    var rowTel = String(data[i][3] || "");
    if (rowName === String(ugyfel) && ((email && rowEmail === String(email)) || (tel && rowTel === String(tel)))) {
      sh.getRange(i + 1, 1, 1, 6).setValues([[ugyfel, cim, email, tel, dateStr, "NEM"]]);
      return;
    }
  }

  sh.appendRow([ugyfel, cim, email, tel, dateStr, "NEM"]);
}

function doGet(e) {
  try {
    var p = (e && e.parameter) ? e.parameter : {};

    if (p.mode === 'ai_parse' || p.text || p.text_content) {
      var txt = p.text || p.text_content || "";
      return jsonResponse(JSON.parse(aiParse(txt)));
    }

    if (validMode(p.mode) && (p.ugyfel || p.nev)) {
      return doPost(e);
    }

    return jsonResponse({ status: "ok", message: "Northwind ULTRA Kapu aktív" });
  } catch (err) {
    return jsonResponse({ status: "error", message: err.toString() });
  }
}

function aiParse(text) {
  var apiKey = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY');
  if (!apiKey) {
    return JSON.stringify({ error: 'Hiányzik a GEMINI_API_KEY a Script Properties-ből.' });
  }

  var apiUrl = "https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=" + apiKey;
  var prompt =
    'Vedd ki az alábbi magyar szövegből az adatokat, és adj vissza szigorúan JSON-t ezekkel a mezőkkel: nev, cim, tel, email, tipus. ' +
    'A tipus csak ez lehet: TELEPÍTÉS, JAVÍTÁS, FELMÉRÉS, KARBANTARTÁS, EGYÉB. ' +
    'Ha valami hiányzik, legyen üres string. Csak JSON-t adj vissza. Szöveg: "' +
    String(text || '').replace(/"/g, '\\"') + '"';

  var payload = {
    contents: [{ parts: [{ text: prompt }] }],
    generationConfig: {
      temperature: 0.1,
      response_mime_type: "application/json"
    }
  };

  try {
    var response = UrlFetchApp.fetch(apiUrl, {
      method: "post",
      contentType: "application/json",
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    });

    var status = response.getResponseCode();
    var body = response.getContentText();

    if (status < 200 || status >= 300) {
      return JSON.stringify({ error: "Gemini API hiba", status: status, body: body });
    }

    var json = JSON.parse(body);
    var raw = (
      json && json.candidates && json.candidates[0] &&
      json.candidates[0].content && json.candidates[0].content.parts &&
      json.candidates[0].content.parts[0] && json.candidates[0].content.parts[0].text
    ) ? json.candidates[0].content.parts[0].text : '{}';

    try {
      var parsed = JSON.parse(raw);
      return JSON.stringify({
        nev: parsed.nev || "",
        cim: parsed.cim || "",
        tel: parsed.tel || "",
        email: parsed.email || "",
        tipus: parsed.tipus || ""
      });
    } catch (e2) {
      return JSON.stringify({ error: "A Gemini válasza nem volt érvényes JSON.", raw: raw });
    }
  } catch (e) {
    return JSON.stringify({ error: e.toString() });
  }
}

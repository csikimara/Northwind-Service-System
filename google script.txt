// 4. EMLÉKEZTETŐ TÁBLÁZAT FRISSÍTÉSE - VÁLTOZATLAN
function frissitEmlekezteto(ugyfel, cim, email, tel, honap) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Emlékeztetők") || ss.insertSheet("Emlékeztetők");
  if (sheet.getLastRow() === 0) sheet.appendRow(["Ügyfél", "Cím", "Email", "Telefon", "Következő Dátum", "Értesítve"]);
  
  var nextDate = new Date();
  nextDate.setMonth(nextDate.getMonth() + parseInt(honap));
  var dateStr = Utilities.formatDate(nextDate, Session.getScriptTimeZone(), "yyyy-MM-dd");
  sheet.appendRow([ugyfel, cim, email, tel, dateStr, "NEM"]);
}

// 1. ESTI ÖSSZEFOGLALÓ (Naptár + Időjárás + Hírek)
// ============================================================
function estiJelentes() {
  var holnap = new Date();
  holnap.setDate(holnap.getDate() + 1);
  var datumString = Utilities.formatDate(holnap, Session.getScriptTimeZone(), "yyyy-MM-dd");
  
  // 1. NAPTÁR ADATOK
  var naptar = CalendarApp.getDefaultCalendar();
  var esemenyek = naptar.getEventsForDay(holnap);
  var menetrend = "📅 HOLNAPI MUNKÁID (" + datumString + "):\n";
  
  if (esemenyek.length > 0) {
    esemenyek.sort(function(a, b) { return a.getStartTime() - b.getStartTime(); });
    esemenyek.forEach(function(e) {
      var ido = Utilities.formatDate(e.getStartTime(), Session.getScriptTimeZone(), "HH:mm");
      menetrend += "👉 " + ido + " - " + e.getTitle() + " (" + (e.getLocation() || "Nincs helyszín") + ")\n";
    });
  } else {
    menetrend += "Nincs rögzített munka holnapra.\n";
  }

  // 2. IDŐJÁRÁS - Betonbiztos megoldás (Open-Meteo)
  var idojaras = "\n🌤️ IDŐJÁRÁS (Budapest):\n";
  try {
    // Ez a szerver tiszta adatokat küld, nem fog "HTML kód" érkezni
    var url = "https://api.open-meteo.com/v1/forecast?latitude=47.4984&longitude=19.0405&current_weather=true&timezone=Europe%2FBerlin";
    var res = UrlFetchApp.fetch(url);
    var json = JSON.parse(res.getContentText());
    
    if (json.current_weather) {
      var temp = json.current_weather.temperature;
      var szel = json.current_weather.windspeed;
      var kod = json.current_weather.weathercode;
      
      // Időjárás kód fordítása egyszerű szövegre
      var leiras = "Tiszta idő";
      if (kod > 0) leiras = "Részben felhős";
      if (kod > 50) leiras = "Eső várható";
      if (kod > 70) leiras = "Havazás";
      
      idojaras += "🌡️ Hőmérséklet: " + temp + "°C\n";
      idojaras += "💨 Szél: " + szel + " km/h\n";
      idojaras += "📝 Állapot: " + leiras;
    }
  } catch(e) {
    idojaras += "Az időjárás adatok jelenleg nem érhetőek el a szerverről.";
  }
  // 3. HÍREK / FORGALOM
  var forgalom = "\n⚠️ VÁROSI AKADÁLYOK:\n";
  try {
    var news = UrlFetchApp.fetch("https://index.hu/24ora/rss/").getContentText();
    if (news.indexOf("lezárják") > -1 || news.indexOf("korlátozás") > -1 || news.indexOf("lezárás") > -1) {
      forgalom += "FIGYELEM! Lezárások vagy korlátozások várhatók Budapesten!";
    } else {
      forgalom += "Nincs hír rendkívüli forgalmi akadályról.";
    }
  } catch(e) { 
    Logger.log("Hírek hiba: " + e.toString());
    forgalom += "Hírek jelenleg nem elérhetők."; 
  }

  // 4. KÜLDÉS
  var targy = "📋 Northwind Esti Jelentés - " + datumString;
  var teljesSzoveg = menetrend + idojaras + forgalom + "\n\nÜdv, Gemini & Northwind Bot";

  try {
    MailApp.sendEmail("northwindhutestechnikakft@gmail.com", targy, teljesSzoveg);
    Logger.log("Jelentés elküldve a Gmail címre.");
  } catch(e) { 
    Logger.log("Küldési hiba: " + e.message); 
  }
}

  // Próbáljuk kiszedni a nevet, emailt, telefont a levél szerkezetéből
  let nev = text.match(/Név:\s*(.*)/)?.[1] || "Demecs Zsolt";
  let email = text.match(/Email:\s*(.*)/)?.[1] || "zsolt.demecs@gmail.com";
  let tel = text.match(/Telefon:\s*(.*)/)?.[1] || "+36302113113";
  
  // Keresünk kulcsszavakat a szövegben
  let fal = text.toLowerCase().includes("tégla") ? "tégla" : (text.toLowerCase().includes("beton") ? "vasbeton" : "tégla");
  let gep = text.includes("Fujitsu") ? "Fujitsu Eco ASYG09/12KPCA" : "Ismeretlen";
  let cso = text.match(/(\d+)\s*méter/)?.[1] || "6";

  let eredmeny = {
    "ugyfel": nev.trim(),
    "cim": "12. kerület (levélből)",
    "email": email.trim(),
    "tel": tel.trim(),
    "gep_tipus": gep,
    "rez_hossz": cso,
    "fal_tipusa": fal,
    "megjegyzes": "AUTOMATA BEOLVASÁS: " + text.substring(0, 150) + "..."
  };

  return JSON.stringify(eredmeny, null, 2);
}

// ============================================================
function doGet(e) {
  try {
    var p = e.parameter;
    
    // Ha nincs megadva mód, küldjön egy alap üzenetet
    if (!p || !p.mode) {
      return ContentService.createTextOutput("A Northwind API él és mozog. Mód hiányzik.");
    }

    // Gmail beolvasás mód
    
    // A doGet függvény vége felé a return-öket cseréld erre:
if (p.mode === 'gmail_parse') {
  var eredmeny = gmailSearchAndParse(p.text);
  // EZZEL AZ ÚJ SORRAL ENGEDÉLYEZZÜK A BÖNGÉSZŐNEK A BEOLVASÁST:
  return ContentService.createTextOutput(eredmeny)
    .setMimeType(ContentService.MimeType.JSON)
    .setHeader('Access-Control-Allow-Origin', '*'); 
}
    
    // AI beolvasás mód (szövegből)
    if (p.mode === 'ai_parse') {
      var eredmeny = aiParse(p.text);
      return ContentService.createTextOutput(eredmeny).setMimeType(ContentService.MimeType.JSON);
    }
    
    return ContentService.createTextOutput("Ismeretlen mód: " + p.mode);
    
  } catch (hiba) {
    return ContentService.createTextOutput("Hiba a kapuban: " + hiba.toString());
  }
}
